# Feasibility spike: React Native 0.88 ReactCommon (cxx platform) + Hermes V1 on Linux x86_64.
cmake_minimum_required(VERSION 3.13)
project(oxy_rn_spike C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Several React Native headers use <cstdint> types without including it; libc++ (Android, Apple)
# includes it transitively, libstdc++ does not.
add_compile_options(-include cstdint)
foreach(var RN_DIR HERMES_SRC HERMES_BUILD CODEGEN_DIR THIRD_PARTY_DIR)
  if(NOT DEFINED ${var})
    message(FATAL_ERROR "${var} is required; run build.sh")
  endif()
endforeach()
set(REACT_COMMON_DIR ${RN_DIR}/ReactCommon)
set(REACT_ANDROID_DIR ${RN_DIR}/ReactAndroid)

# React Native's components link `yoga`; ReactAndroid defines it in yogajni (JNI). Off Android
# it is the core library.
add_library(yoga INTERFACE)
target_link_libraries(yoga INTERFACE yogacore)

# Android's liblog has no Linux counterpart; glog writes to stderr off Android.
add_library(log INTERFACE)

add_library(hermes-engine::hermesvm SHARED IMPORTED GLOBAL)
set_target_properties(hermes-engine::hermesvm PROPERTIES
  IMPORTED_LOCATION ${HERMES_BUILD}/lib/libhermesvm.so
  INTERFACE_INCLUDE_DIRECTORIES "${HERMES_SRC}/API;${HERMES_SRC}/public;${HERMES_BUILD}/lib/config")

foreach(dir glog boost double-conversion fast_float fmt folly)
  add_subdirectory(${THIRD_PARTY_DIR}/${dir} third-party/${dir})
endforeach()
# folly's libstdc++ exception internals name std::unexpected_handler, deprecated in C++20.
target_compile_options(folly_runtime PRIVATE -Wno-deprecated-declarations)
add_subdirectory(${REACT_ANDROID_DIR}/src/main/jni/first-party/fbgloginit fbgloginit)

# Generated by @react-native/codegen from react-native/src (see codegen.sh); the Android-only
# Java/JNI glue (FBReactNativeSpec-generated.cpp) is not built.
set(CODEGEN_DIR ${CODEGEN_DIR})
file(GLOB rncore_SRC CONFIGURE_DEPENDS ${CODEGEN_DIR}/react/renderer/components/FBReactNativeSpec/*.cpp)
add_library(react_codegen_rncore OBJECT ${rncore_SRC})
target_include_directories(react_codegen_rncore PUBLIC ${CODEGEN_DIR} ${CODEGEN_DIR}/react/renderer/components
  ${CODEGEN_DIR}/react/renderer/components/FBReactNativeSpec)
target_link_libraries(react_codegen_rncore jsi folly_runtime glog react_debug react_nativemodule_core
  react_renderer_componentregistry react_renderer_core react_renderer_debug react_renderer_graphics
  react_renderer_imagemanager react_renderer_mapbuffer react_utils rrc_image rrc_view yoga)
target_compile_options(react_codegen_rncore PRIVATE -std=c++20 -fexceptions -frtti -DFOLLY_NO_CONFIG=1 -DHERMES_V1_ENABLED=1)

# Like ReactAndroid's libreactnative.so: every ReactCommon OBJECT library in one shared library.
function(collect_targets directory out)
  get_property(targets DIRECTORY ${directory} PROPERTY BUILDSYSTEM_TARGETS)
  get_property(children DIRECTORY ${directory} PROPERTY SUBDIRECTORIES)
  foreach(child ${children})
    collect_targets(${child} nested)
    list(APPEND targets ${nested})
  endforeach()
  set(${out} ${targets} PARENT_SCOPE)
endfunction()
collect_targets(${CMAKE_CURRENT_SOURCE_DIR} all_targets)
set(reactnative_OBJECTS)
foreach(target ${all_targets})
  get_target_property(type ${target} TYPE)
  if(type STREQUAL "OBJECT_LIBRARY")
    list(APPEND reactnative_OBJECTS $<TARGET_OBJECTS:${target}>)
    list(APPEND reactnative_LINKS ${target})
  endif()
endforeach()
add_library(reactnative SHARED ${reactnative_OBJECTS})
target_link_libraries(reactnative PUBLIC ${reactnative_LINKS} hermes-engine::hermesvm folly_runtime glog fmt
  double-conversion yogacore atomic)

add_executable(oxy-rn-spike host.cpp)
target_link_libraries(oxy-rn-spike reactnative)
target_compile_options(oxy-rn-spike PRIVATE -std=c++20 -fexceptions -frtti -DFOLLY_NO_CONFIG=1 -DHERMES_V1_ENABLED=1)
